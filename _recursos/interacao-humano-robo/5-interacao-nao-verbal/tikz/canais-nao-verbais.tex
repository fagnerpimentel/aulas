  \newcommand{\canal}[3]{%
    % #1 = posição (x,y), #2 = rótulo, #3 = ícone (macro)
    \begin{scope}[shift={(#1)}]
      \fill[dcream, draw=dblue, line width=1pt] (0,0) circle (1.05);
      #3
      \node[font=\small, align=center, text width=3.0cm] at (0,-1.5) {#2};
    \end{scope}}

  % Ícone: olhar (olho + pupila)
  \newcommand{\iconOlhar}{
    \draw[dbody, line width=1pt] (-0.62,0) .. controls (-0.3,0.42) and (0.3,0.42) .. (0.62,0)
                                  .. controls (0.3,-0.42) and (-0.3,-0.42) .. (-0.62,0);
    \fill[dblue] (0,0) circle (0.20);
    \fill[white] (0.06,0.06) circle (0.06);
  }
  % Ícone: gesto (apenas a mão aberta, cinco dedos -- "oi"/"pare")
  \newcommand{\iconGesto}{
    \begin{scope}[shift={(0,-0.15)}]

      \fill[dblue!12, draw=dblue, line width=1pt] (-0.13,-0.62) rectangle (0.13,-0.32);

    \end{scope}
  }
  % Ícone: imitação (duas figuras espelhadas)
  \newcommand{\iconImitacao}{
    \fill[black!12, draw=black!75, line width=0.7pt] (-0.30,0.28) circle (0.16);

  \newcommand{\iconToque}{

    \fill[dorange!25, draw=dorange, line width=0.9pt] (0.38,0) circle (0.30);
