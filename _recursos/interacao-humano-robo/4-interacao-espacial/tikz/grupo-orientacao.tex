  \newcommand{\perso}[3]{
    \begin{scope}[shift={(#1,#2)}, rotate=#3]
      \fill[black!12, draw=black!75, line width=0.8pt] (0,0) circle (0.30);
      \fill[black!75] (0,0.30) -- (-0.14,0.46) -- (0.14,0.46) -- cycle;
    \end{scope}}
  \newcommand{\personA}[2]{% angulo no circulo, raio  -> pessoa virada para o centro
    \pgfmathsetmacro\ppx{#2*cos(#1)}\pgfmathsetmacro\ppy{#2*sin(#1)}\pgfmathsetmacro\ppr{#1+90}
    \perso{\ppx}{\ppy}{\ppr}}
  \newcommand{\personO}[3]{% angulo, raio, inclinacao para fora
    \pgfmathsetmacro\ppx{#2*cos(#1)}\pgfmathsetmacro\ppy{#2*sin(#1)}\pgfmathsetmacro\ppr{#1+90+#3}
    \perso{\ppx}{\ppy}{\ppr}}
  \newcommand{\robo}[3]{
    \begin{scope}[shift={(#1,#2)}, rotate=#3]
      \fill[black!5, draw=black!80, line width=0.9pt, rounded corners=1.5pt] (-0.28,-0.22) rectangle (0.28,0.22);
      \fill[dblue] (0,0.22) -- (-0.14,0.40) -- (0.14,0.40) -- cycle;
      \node[font=\tiny, black!55] at (0,0) {R};
    \end{scope}}
