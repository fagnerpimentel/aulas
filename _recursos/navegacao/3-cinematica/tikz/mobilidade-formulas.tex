\begin{tikzpicture}[x=1cm,y=1cm]

  \def\U{1.7}   % 1 grau de liberdade = 1.7 cm
  \def\H{0.34}  % meia-altura da barra
  \def\LX{-0.15} % fim da area de rotulo (x=0 do eixo)

  % ---- eixo 0..3 ----
  \draw[black!55, line width=0.8pt] (0,0.55) -- ({3*\U+0.35},0.55);
  \foreach \k in {0,1,2,3}{
    \draw[black!55, line width=0.8pt] ({\k*\U},0.48) -- ({\k*\U},0.62);
    \node[dsub, font=\scriptsize] at ({\k*\U},0.85) {\k};
  }
  \node[dsub, font=\scriptsize, anchor=west] at ({3*\U+0.5},0.85)
     {graus que o robô controla ($\delta_M$)};

  % ---- linhas: (nome, dm, ds, y) ----
  \foreach \nome/\dm/\ds/\yy [count=\i] in {%
      Diferencial/2/0/0,
      {Triciclo / bicicleta}/1/1/-1,
      Omnidirecional/3/0/-2,
      Omni-steer/2/1/-3,
      Two-steer/1/2/-4}{
    \node[font=\scriptsize, anchor=east] at (\LX,\yy) {\nome};
    % delta_m (direto)
    \fill[dblue!75] (0,{\yy-\H}) rectangle ({\dm*\U},{\yy+\H});
    % delta_s (esterco) -- empilhado
    \fill[dorange!80] ({\dm*\U},{\yy-\H}) rectangle ({(\dm+\ds)*\U},{\yy+\H});
    \draw[black!45, line width=0.5pt] (0,{\yy-\H}) rectangle ({(\dm+\ds)*\U},{\yy+\H});
    % rotulo numerico
    \pgfmathtruncatemacro{\dM}{\dm+\ds}
    \node[dsub, font=\scriptsize, anchor=west] at ({(\dm+\ds)*\U+0.25},\yy)
       {$\delta_m{+}\delta_s = \dm{+}\ds = \dM$};
  }

  % ---- legenda ----
  \fill[dblue!75] (\LX-2.2,-5.15) rectangle ++(0.3,0.3);
  \node[font=\scriptsize, anchor=west] at (\LX-1.8,-5.0)
     {$\delta_m$ --- \textbf{direto}: só a velocidade das rodas};
  \fill[dorange!80] ({3*\U-1.2},-5.15) rectangle ++(0.3,0.3);
  \node[font=\scriptsize, anchor=west] at ({3*\U-0.8},-5.0)
     {$\delta_s$ --- \textbf{indireto}: esterçar e então mover};

\end{tikzpicture}
