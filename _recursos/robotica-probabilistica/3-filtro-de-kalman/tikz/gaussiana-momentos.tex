  \node[black!70, font=\small] at (5.6,4.6)
     {\textbf{representação de momentos:} média $\mu$ e covariância $\Sigma$};

  % ---------- painel 1D ----------
  \begin{scope}
    \def\mnx{2.4}\def\snx{0.9}
    \draw[daxis] (-0.2,0) -- (5.2,0) node[right, dsub] {$x$};
    \draw[daxis] (0,-0.2) -- (0,3.6);
    \draw[dcurve, samples=120, domain=-0.2:5.2, smooth, variable=\x]
      plot (\x, {2.7*2.5066*gauss(\x,\mnx,\snx)});
    \draw[dorange, line width=1pt, dashed] (\mnx,0) -- (\mnx,{2.7*2.5066*gauss(\mnx,\mnx,\snx)});
    \node[dorange, font=\small] at (\mnx,-0.42) {$\mu$};
    \draw[<->, dgreen, line width=0.9pt] (\mnx,0.85) -- ++(\snx,0)
        node[midway, above, dgreen, font=\footnotesize] {$\Sigma$};
    \node[dlbl] at (2.4,-1.2) {caso 1D: $\mathcal{N}(\mu,\Sigma^2)$};
  \end{scope}

  % ---------- painel 2D (elipse de covariância) ----------
  \begin{scope}[shift={(7.6,0)}]
    \draw[daxis] (-0.2,0) -- (4.4,0) node[right, dsub] {$x_1$};
    \draw[daxis] (0,-0.2) -- (0,3.6) node[above, dsub] {$x_2$};
    \fill[dblue!10] (2.2,1.8) ellipse [x radius=1.5, y radius=0.95, rotate=28];
    \draw[dblue!35, line width=0.8pt] (2.2,1.8) ellipse [x radius=1.5, y radius=0.95, rotate=28];
    \draw[dblue, line width=1pt] (2.2,1.8) ellipse [x radius=1.0, y radius=0.62, rotate=28];
    \fill[dorange] (2.2,1.8) circle (1.6pt);
    \node[dorange, font=\small, anchor=north west] at (2.35,1.7) {$\boldsymbol\mu$};
    \draw[->, dgreen, line width=0.9pt] (2.2,1.8) -- ++(28:1.0)
        node[midway, above left=-3pt, dgreen, font=\footnotesize] {$\boldsymbol\Sigma$};
    \node[dlbl] at (2.1,-1.2) {caso $n$D: elipsoide de $\boldsymbol\Sigma$};
  \end{scope}

\end{tikzpicture}
