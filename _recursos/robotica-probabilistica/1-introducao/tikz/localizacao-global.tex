\documentclass[border=3pt]{standalone}
\input{preamble}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

  \def\dA{2.3}\def\dB{5.6}\def\dC{9.2}
  \def\W{11}
  \def\dy{4.7}

  \newcommand{\scene}[2]{% #1 = rótulo, #2 = x do robô (<=0 = sem robô)
    \fill[dshade!70] (0,2.35) rectangle (\W,3.15);
    \draw[black!55, line width=0.7pt] (0,2.35) rectangle (\W,3.15);
    \foreach \d in {\dA,\dB,\dC}{
      \fill[dorange!35] (\d-0.4,2.35) rectangle (\d+0.4,3.15);
      \draw[black!55, line width=0.6pt] (\d-0.4,2.35) rectangle (\d+0.4,3.15);
    }
    \ifdim #2pt>0pt \fill[dred] (#2,2.1) circle (2.3pt); \fi
    \draw[daxis] (-0.15,0) -- (\W+0.3,0) node[right, dsub] {$x$};
    \node[black!60, font=\footnotesize, anchor=west] at (0.0,3.55) {#1};
    \node[dsub, anchor=east] at (-0.25,1.2) {$bel(x)$};
  }

  % (a) prior uniforme
  \begin{scope}
    \scene{(a) crença inicial: uniforme (incerteza máxima)}{-1}
    \draw[dblue, line width=1.4pt] (0.1,0.6) -- (\W-0.1,0.6);
  \end{scope}

  % (b) percebe "porta": 3 picos
  \begin{scope}[shift={(0,-\dy)}]
    \scene{(b) percebe ``porta'' $\to$ 3 picos (portas indistinguíveis)}{\dA}
    \draw[dblue, line width=1.4pt, samples=240, domain=0:\W, smooth, variable=\x]
      plot (\x, {0.12 + 1.38*(gauss(\x,\dA,0.34)+gauss(\x,\dB,0.34)+gauss(\x,\dC,0.34))});
  \end{scope}

  % (c) após o movimento: deslocada e suavizada
  \begin{scope}[shift={(0,-2*\dy)}]
    \scene{(c) após o movimento: deslocada e suavizada}{\dB}
    \draw[dblue, line width=1.4pt, samples=240, domain=0:\W, smooth, variable=\x]
      plot (\x, {0.12 + 1.50*(gauss(\x,\dA+1.3,0.60)+gauss(\x,\dB+1.3,0.60)+gauss(\x,\dC+1.3,0.60))});
  \end{scope}

  % (d) percebe "porta" de novo: um pico dominante
  \begin{scope}[shift={(0,-3*\dy)}]
    \scene{(d) percebe ``porta'' de novo: robô confiante}{\dB}
    \draw[dblue, line width=1.4pt, samples=260, domain=0:\W, smooth, variable=\x]
      plot (\x, {0.10 + 1.58*gauss(\x,\dB,0.34) + 0.40*gauss(\x,\dA,0.40) + 0.38*gauss(\x,\dC,0.40)});
  \end{scope}

\end{tikzpicture}
\end{document}
